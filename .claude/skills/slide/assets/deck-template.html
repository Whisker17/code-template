from trl import GRPOTrainer

def tests_pass(completions, **kw):
    # fraction of asserts the code passes
    return [run(c) for c in completions]

trainer = GRPOTrainer(model="Qwen/Qwen3-0.6B", reward_funcs=tests_pass)
trainer.train()
